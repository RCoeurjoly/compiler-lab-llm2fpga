# Immutable Series-7 ROM mapping using native LUTRAM geometry.
# Matches Yosys 0.66 xilinx/lutrams_xc5v.txt's SDP geometry and native map.
# Only packed RAM32M/RAM64M: this nextpnr backend rejects single-bit SP ROMs.
# Deliberately omits prune_rom: opt_mem removes inactive write ports, but
# these selected model ROMs must remain eligible for initialized LUTRAM.
# Does not change primitive behavior or the installed Yosys libraries.
ram distributed $__XILINX_LUTRAM_SDP_ {
 cost 8;
 widthscale 7;
 option "ABITS" 5 { abits 5; widths 6 global; }
 option "ABITS" 6 { abits 6; widths 3 global; }
 init no_undef;
 port sw "W" { clock posedge; }
 port ar "R" { }
}
